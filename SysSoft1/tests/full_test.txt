def loops(a of int)
    while a > 0
        a = a - 1;
    end

    until a == 10
        a = a + 1;
    end

    while a > 0
        if a == 5 then break;
        a = a - 1;
    end

    end

    end
end

def combined(data of int array[10], n of int)
    begin
        def helper(x of int) of int
            x * 2;
        end

        while n > 0
            if n == 5 then break;
            data[n] = helper(n);
            n = n - 1;
        end

        until n == 10
            n = n + 1;
        end

        data[0..n];
    end
end
